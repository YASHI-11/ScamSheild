# ScamShield dependencies
#
# ON IBM LinuxONE (s390x), DO NOT USE THIS FILE WITH pip.
# ----------------------------------------------------------------------
# There are no s390x wheels on PyPI for numpy/scikit-learn/xgboost, so pip
# falls back to compiling them from source. Use the OS packages instead:
#
#   Bare Ubuntu LinuxONE guest (verified on Ubuntu 24.04 / s390x):
#     sudo apt-get install -y python3-numpy python3-pandas python3-sklearn \
#                             python3-xgboost python3-joblib \
#                             python3-matplotlib python3-pytest
#     (Ubuntu 24.04 also refuses `pip install --user` outright - PEP 668.)
#
#   Marist JupyterLab image (l1cc/jupyterlab-image-s390x):
#     already has all of the above; only `pip install --user streamlit plotly`
#
# Streamlit + plotly are NOT installable on a bare s390x box: streamlit
# requires pyarrow, which publishes no s390x wheel and is not in the Ubuntu
# archive. See README section 4 for the split-demo workaround.
#
# This file exists so the project installs cleanly on a laptop and so the
# judges can see exactly what it depends on.

# --- core: pre-installed on LinuxONE, install locally -----------------
pandas>=2.0
numpy>=1.24
scikit-learn>=1.3
xgboost>=2.0
joblib>=1.3

# --- dashboard: pip install these inside the container ----------------
streamlit>=1.30
plotly>=5.18

# --- the precision/recall PNG (optional) ------------------------------
# train.py always writes artifacts/pr_curve.csv; the PNG is written only if
# matplotlib is importable, so this is a nice-to-have, not a requirement.
matplotlib>=3.7

# --- tests ------------------------------------------------------------
pytest>=7.4
